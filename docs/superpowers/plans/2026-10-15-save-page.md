# Save a Page to Draftab Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Clicking Draftab's toolbar icon saves the page's main text as a new note; right-click → "Save to Draftab" on a selection saves just the selection. Either way a new Draftab tab opens with the note.

**Architecture:** The service worker injects a capture script into the active tab (`activeTab` + `scripting`), stores the result as a temporary `clip:<id>` key and opens `newtab.html?clip=<id>`. The new tab (which has a DOM) converts the HTML into Draftab blocks with the editor's ProseMirror schema, saves the note and opens it.

**Tech Stack:** TypeScript, Vite, Tiptap 3 / ProseMirror (`@tiptap/pm/model`), `@mozilla/readability`, Vitest + `happy-dom` (tests only).

**Spec:** `docs/superpowers/specs/2026-10-15-save-page-design.md`

## Global Constraints
- Strict TypeScript; no UI framework; direct, readable code (the user reads basic JS).
- All `chrome.storage` access goes through `src/storage.ts`.
- The Tiptap JSON is the source of truth; markdown only on export.
- Permissions after this change: `storage`, `unlimitedStorage`, `alarms`, `downloads`, `activeTab`, `scripting`, `contextMenus`. Nothing else.
- Every dependency shipped is MIT, except `@mozilla/readability` (Apache 2.0), recorded in SPEC.md. `happy-dom` is dev-only.
- Everything in English (code, comments, UI text, docs); every UI text exists in both `EN` and `PT` in `src/i18n.ts`.
- SPEC.md changes before code. Replies to the user in Brazilian Portuguese.
- Commits end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

## Review Focus
1. **Clicking twice quickly** (two captures before the first tab loads): both notes must be created. Cleanup removes only clips older than 10 minutes (test in Task 3).
2. **Reloading the tab that opened a clipped note** must not create a second note: `takeClip` deletes the key and `?clip=` leaves the address (test in Task 3, check in Task 5).
3. **`h4`–`h6` in the content** must become `###` headings, not lose their heading-ness or text (test in Task 2).
4. **Hostile or junk HTML** (`<script>`, `<style>`, `javascript:` links, inline event handlers) must never run in the extension page or end up as text or links in the note. The conversion parses with `DOMParser` (an inert document, nothing loads or runs) (tests in Task 2).
5. **A selection with no text** (only an image) must show the "no main text" message instead of creating an empty note (test in Task 2).

import type { JSONContent } from '@tiptap/core'
import { DOMParser as SchemaParser } from '@tiptap/pm/model'
import { contentSchema } from './extensions'
import { newBlankNote, textOf, withUpdatedTitle, type Note, type NoteFont } from './notes'

// Saving web pages (see "Saving web pages" in SPEC.md): what the capture in the page
// hands to the new tab, and how it becomes a note.

export type ClipError = 'unavailable' | 'empty' // a page extensions can't run on / no main text
export type PageClip = { title: string; url: string; html: string }
export type Clip = PageClip | { error: ClipError }

// Elements whose content never goes into the note
const DROPPED =
  'script, style, noscript, template, img, picture, video, audio, iframe, object, embed, svg, canvas, math, form, button, input, select, textarea'

// A new note: the page title, a link to the page, then the content.
// null when the content has no text (e.g. a selection with only an image).
// textSize and font: the "Text size of new notes" and "Font of new notes" settings
export function clipToNote(clip: PageClip, textSize = 100, font: NoteFont = 'sans'): Note | null {
  const content = htmlToBlocks(clip.html, clip.url)
  if (textOf({ type: 'doc', content }).trim() === '') return null

  const title = clip.title.replace(/\s+/g, ' ').trim() || domainOf(clip.url)
  const doc: JSONContent = {
    type: 'doc',
    content: [
      { type: 'heading', attrs: { level: 1 }, content: [{ type: 'text', text: title }] },
      { type: 'paragraph', content: [{ type: 'text', text: clip.url, marks: [{ type: 'link', attrs: { href: clip.url } }] }] },
      ...content,
    ],
  }
  return withUpdatedTitle({ ...newBlankNote(textSize, font), doc })
}

// Draftab blocks for a piece of HTML. baseUrl: the page's address, for relative links.
export function htmlToBlocks(html: string, baseUrl: string): JSONContent[] {
  // DOMParser makes an inert document: no script runs and nothing (images…) loads
  const page = new DOMParser().parseFromString(html, 'text/html')
  const body = page.body

  for (const el of body.querySelectorAll(DROPPED)) el.remove()

  // The page title is the note's only level 1: h1/h2 → level 2, h3-h6 → level 3
  for (const old of body.querySelectorAll('h1, h2, h3, h4, h5, h6')) {
    const heading = page.createElement(Number(old.tagName[1]) <= 2 ? 'h2' : 'h3')
    heading.append(...old.childNodes)
    old.replaceWith(heading)
  }

  // A table becomes one paragraph per row, cells separated by " · "
  for (const table of body.querySelectorAll('table')) {
    const rows = [...table.querySelectorAll('tr')].map((row) => {
      const cells = [...row.children].filter((cell) => cell.tagName === 'TD' || cell.tagName === 'TH')
      const paragraph = page.createElement('p')
      paragraph.textContent = cells.map((cell) => cell.textContent?.trim()).filter(Boolean).join(' · ')
      return paragraph
    })
    table.replaceWith(...rows)
  }

  // Links: absolute addresses, only http, https and mailto; others keep just their text
  for (const link of body.querySelectorAll('a')) {
    const href = absoluteUrl(link.getAttribute('href'), baseUrl)
    if (href && /^(https?|mailto):/i.test(href)) link.setAttribute('href', href)
    else link.replaceWith(...link.childNodes)
  }

  // The editor's schema keeps only Draftab blocks and marks
  const doc = SchemaParser.fromSchema(contentSchema()).parse(body).toJSON() as JSONContent
  return (doc.content ?? []).filter((block) => !(block.type === 'paragraph' && textOf(block).trim() === ''))
}

function absoluteUrl(href: string | null, baseUrl: string): string | null {
  if (href === null) return null
  try {
    return new URL(href, baseUrl).href
  } catch {
    return null
  }
}

// "example.com" for https://example.com/post; the address itself when it has no domain
function domainOf(url: string): string {
  try {
    return new URL(url).hostname || url
  } catch {
    return url
  }
}
```

// ---------- Captured pages ----------

// A page captured by the service worker waits here for the new tab that turns it into
// a note (see "Saving web pages" in SPEC.md). Not a note: never listed or backed up.
const CLIP_PREFIX = 'clip:'
const CLIP_MAX_AGE = 10 * 60 * 1000 // 10 minutes

export async function saveClip(clip: Clip): Promise<string> {
  const id = crypto.randomUUID()
  await chrome.storage.local.set({ [CLIP_PREFIX + id]: { ...clip, createdAt: Date.now() } })
  return id
}

// Reads the clip and deletes it, so reloading the tab doesn't create the note again
export async function takeClip(id: string): Promise<Clip | null> {
  const key = CLIP_PREFIX + id
  const result = await chrome.storage.local.get(key)
  const stored = result[key] as (Clip & { createdAt: number }) | undefined
  if (!stored) return null
  await chrome.storage.local.remove(key)
  const { createdAt, ...clip } = stored
  return clip as Clip
}

// Clips whose tab never opened them (closed too soon). Only old ones, so two quick
// captures don't delete each other.
export async function deleteOldClips(now = Date.now()): Promise<void> {
  const everything = await chrome.storage.local.get(null)
  const old = Object.entries(everything)
    .filter(([key, value]) => key.startsWith(CLIP_PREFIX) && ((value as { createdAt?: number }).createdAt ?? 0) < now - CLIP_MAX_AGE)
    .map(([key]) => key)
  if (old.length > 0) await chrome.storage.local.remove(old)
}
```

import type { PageClip } from './clip'

// Injected into the page when the toolbar icon is clicked (see background.ts).
// It runs in the extension's isolated world: the page can't see or call it.
// Readability changes the document it reads, so it gets a copy.
function capturePage(): PageClip | null {
  const article = new Readability(document.cloneNode(true) as Document).parse()
  if (!article?.content || !article.textContent?.trim()) return null
  return { title: document.title, url: location.href, html: article.content }
}

;(globalThis as { draftabCapturePage?: () => PageClip | null }).draftabCapturePage = capturePage

// Runs the capture; a page extensions can't run on gives "unavailable", no text gives "empty"
async function capture(
  tabId: number,
  frameId: number | undefined,
  run: (target: chrome.scripting.InjectionTarget) => Promise<PageClip | null>,
): Promise<Clip> {
  const target = frameId === undefined ? { tabId } : { tabId, frameIds: [frameId] }
  try {
    return (await run(target)) ?? { error: 'empty' }
  } catch (error) {
    console.warn('Draftab: capture failed', error)
    return { error: 'unavailable' }
  }
}

// Hands the capture to a new Draftab tab, which turns it into a note
async function openClip(clip: Clip) {
  await deleteOldClips()
  const id = await saveClip(clip)
  await chrome.tabs.create({ url: `newtab.html?clip=${id}` })
}

// Injected into the page: the HTML of the selection. Must not use anything from outside
// this function (Chrome copies only its code into the page).
function captureSelection(): { title: string; url: string; html: string } | null {
  const selection = getSelection()
  if (!selection || selection.isCollapsed) return null
  const box = document.createElement('div')
  for (let i = 0; i < selection.rangeCount; i++) box.append(selection.getRangeAt(i).cloneContents())
  return { title: document.title, url: location.href, html: box.innerHTML }
}

// The tab was opened by the toolbar icon or the right-click menu, with ?clip=<id>.
// Returns true when it opened a note or showed a message.
async function openClip(id: string): Promise<boolean> {
  history.replaceState(null, '', location.pathname) // a reload doesn't create the note again
  const clip = await takeClip(id)
  if (!clip) return false
  const note = 'error' in clip ? null : clipToNote(clip, newNoteTextSize, newNoteFont)
  if (!note) {
    open(newNote(), false)
    const message = 'error' in clip && clip.error === 'unavailable' ? t('clipUnavailable') : t('clipEmpty')
    setTimeout(() => alert(message), 100) // after the page shows up
    return true
  }
  await saveNote(note)
  updateNoteInList(note.id, note)
  open(note, true)
  return true
}
